// Description:
// Purpose: Find PowerShell processes whose *own* command line (ProcessCommandLine) contains stealth/evasion/automation flags.
// Why run this: Attackers and red teams frequently launch PowerShell with flags that reduce visibility or enable scripted payloads
//   (examples: "-WindowStyle Hidden" to hide a window, "-NoProfile" to bypass user profiles, "-EncodedCommand" to hide payload text).
// When to run: Run anytime you suspect scripted execution, stealthy automation, or when you see unusual process trees referencing PowerShell.
// What to expect from the lab harness: hidden PowerShell child and an encoded-command child created by the harness; you'll see the flags in ProcessCommandLine.
// False positives: legitimate admin tooling, some scheduled automation, and management scripts that intentionally use -NoProfile or -EncodedCommand. Check parent process and account.
// Follow-ups:
//  1) Inspect InitiatingProcessFileName and InitiatingProcessCommandLine to see what launched PowerShell.
//  2) Get process hashes (InitiatingProcessFileHash) and check signatures or TI.
//  3) If you see EncodedCommand, attempt to decode or collect the command for triage (export process data / perform live capture).
let window = 4h;
DeviceProcessEvents
| where Timestamp >= ago(window)
| where FileName in~ ("powershell.exe","pwsh.exe")
| where ProcessCommandLine has_any ("-WindowStyle Hidden","-NoProfile","-EncodedCommand","-e ","-nop")
| project Timestamp, DeviceName, FileName, ProcessCommandLine, InitiatingProcessFileName, InitiatingProcessCommandLine, AccountName, InitiatingProcessFileHash
| order by Timestamp desc


// Description:
// Purpose: Inventory of executables started in the window — aggregate view showing which binaries dominated activity.
// Why run this: Gives a quick, prioritized list of what actually ran during the timeframe so you can spot unusual or new binaries without hard-coding names.
// When to run: Good as a first-step after any simulation or suspected compromise to enumerate the primary executables that generated telemetry.
// What to expect from the lab harness: notepad/mspaint/calc and powershell should be near the top; notepad may appear both as an initiator and as a FileName (if launched to open temp files).
// False positives: installers, legitimate scheduled jobs, and update processes often appear with high counts; combine with Initiators and sample command lines to reduce noise.
// Follow-ups:
//  1) For low-frequency or unfamiliar binaries, investigate file hash, signature, publisher, and parent-child chain.
//  2) Cross-reference with baseline/top-30-day inventory to mark “rare” binaries.
let window = 2h;
DeviceProcessEvents
| where Timestamp >= ago(window)
| summarize
    Count = count(),
    FirstSeen = min(Timestamp),
    LastSeen = max(Timestamp),
    SampleCmdLines = make_set(tostring(ProcessCommandLine), 5),
    SampleInitiators = make_set(tostring(InitiatingProcessFileName), 5)
  by FileName
| order by Count desc


// Description:
// Purpose: Find files written into user temporary folders and the processes that opened them.
// Why run this: Malware and scripts commonly stage payloads or exfiltration artifacts in temp directories; benign tools and installers also use temp — but temporal correlation with suspicious processes matters.
// When to run: Run after detecting new files, process execs, or to validate file-staging behavior in a lab run.
// What to expect from the lab harness: GUID-named temp file(s) and notepad opening those files; you will see the file write events and the notepad process opening them.
// False positives: installers, legitimate applications updating themselves, or benign tooling that stage in temp. Check the initiating process and file contents/hashes.
// Follow-ups:
//  1) Pull file hashes (if available) and submit to TI / file reputation services.
//  2) If suspicious, request file retrieval or endpoint forensic collection (collector) to inspect contents.
//  3) Correlate with DeviceProcessEvents to see who wrote the file and who opened it.
let window = 2h;
DeviceFileEvents
| where Timestamp >= ago(window)
| where FolderPath has @"\AppData\Local\Temp" or FolderPath has @"\Temp\"
| project Timestamp, DeviceName, FileName, FolderPath, ActionType, InitiatingProcessFileName, InitiatingProcessCommandLine
| order by Timestamp desc


// Description:
// Purpose: Detect creation or modification of Run/RunOnce autorun registry values — a common persistence mechanism.
// Why run this: Adversaries commonly create entries under these keys to achieve persistence; defenders and benign installers may also write here, so context matters.
// When to run: Run as part of persistence hunts or after seeing process/file events that suggest eventual re-execution (e.g., staged drop + autorun registration).
// What to expect from the lab harness: an entry named "LabTelemetry" pointing to notepad.exe (or similar) in the Run key under the current user; you'll see the registry write event.
// False positives: legitimate software installers, user-intended startup entries; check the value data and parent process to reduce false positives.
// Follow-ups:
//  1) Inspect RegistryValueData for odd paths, encoded commands, or remote locations.
//  2) Query DeviceProcessEvents around the same timestamp for the initiator process and account.
//  3) If deemed malicious, remove the registry entry and list related persistence artifacts (Scheduled Tasks, Services, Startup folder).
let window = 2h;
DeviceRegistryEvents
| where Timestamp >= ago(window)
| where RegistryKey has @"\Software\Microsoft\Windows\CurrentVersion\Run"
   or RegistryKey has @"\Software\Microsoft\Windows\CurrentVersion\RunOnce"
| project Timestamp, DeviceName, RegistryKey, RegistryValueName, RegistryValueData, InitiatingProcessFileName, InitiatingProcessCommandLine, AccountName
| order by Timestamp desc


// Description:
// Purpose: Find DNS and HTTP probes to external domains — a generic recon / scripted browsing indicator.
// Why run this: Repeated DNS + HTTP to many distinct hosts in a short period is a reliable indicator of discovery or automated recon; attackers and lab scripts both exhibit this pattern.
// When to run: Use this query after seeing suspicious process behavior, or regularly to surface devices performing unusual outbound recon activity.
// What to expect from the lab harness: DNS + HTTP entries for example.com, badssl.com, iana.org, malware-traffic-analysis.net; msftconnecttest.com might fail to resolve depending on environment.
// False positives: normal browsing, update checks, telemetry from benign apps. Combine with InitiatingProcessFileName and frequency to triage (e.g., GUI browser vs hidden headless process).
// Follow-ups:
//  1) Inspect InitiatingProcessFileName and ProcessCommandLine; if those are non-browser or hidden processes, escalate the host for deeper review.
//  2) Compare destination domains against threat intelligence feeds (ThreatIntelligenceIndicator) and flag matches.
//  3) Look for repeating patterns or scheduled recon (use the recon fingerprint query that bins activity).
let window = 2h;
DeviceNetworkEvents
| where Timestamp >= ago(window)
| where ActionType in ("DnsQuery","HttpConnectionInspected","ConnectionAttempted","ConnectionSuccess")
| where RemoteUrl has_any ("example.com","badssl.com","iana.org","malware-traffic-analysis.net","msftconnecttest.com")
| project Timestamp, DeviceName, InitiatingProcessFileName, RemoteUrl, RemoteIP, RemotePort, ActionType
| order by Timestamp desc


// Description:
// Purpose: Spot TCP/connection probes to the RFC5737 TEST-NET address ranges used for documentation (192.0.2.0/24, 198.51.100.0/24, 203.0.113.0/24).
// Why run this: Hits to these IP ranges are very rare in production; their presence typically indicates scripted probes, lab activity, or misconfigured tooling that used example/test addresses.
// When to run: Use when hunting for scripted scanning, sandbox checks, or to identify lab hosts that are running synthetic recon tests.
// What to expect from the lab harness: quick connect attempts to 192.0.2.1, 198.51.100.1, 203.0.113.1; these usually fail to connect but still make detectable network events.
// False positives: extremely unlikely in normal enterprise traffic; if you see many hosts hitting these ranges, validate whether any internal tooling intentionally targets them.
// Follow-ups:
//  1) Correlate with InitiatingProcessFileName to find the script or tool responsible.
//  2) If this is unexpected, isolate the device and capture a process timeline and network session for deeper triage.
//  3) Consider blocking or alerting on these ranges in detection rules for production environments.
let window = 2h;
DeviceNetworkEvents
| where Timestamp >= ago(window)
| where isnotempty(RemoteIP)
| where ipv4_is_in_range(RemoteIP, "192.0.2.0/24")
   or ipv4_is_in_range(RemoteIP, "198.51.100.0/24")
   or ipv4_is_in_range(RemoteIP, "203.0.113.0/24")
| project Timestamp, DeviceName, InitiatingProcessFileName, RemoteIP, RemotePort, ActionType
| order by Timestamp desc


// Description:
// Purpose: Show file writes and process requests related to EICAR artifacts to validate AV/EDR detection behavior.
// Why run this: When intentionally testing detection (EICAR or other test artifacts), pull file creation events, process attempts to open the file, and related ReportId/Alert evidence to see detection lifecycle (create -> detect -> quarantine/upload).
// When to run: Run after a lab harness that writes/executes EICAR, or any AV test. Keep this separate from pre-alert hunts to avoid mixing test artifacts with normal telemetry.
// What to expect from the lab harness: creation of C:\Temp\EICARTest\eicar.com and immediate Defender detection/quarantine events; DeviceFileEvents for the write and possibly for the attempted launch, plus Defender DeviceEvents/alerts.
// False positives: none for EICAR itself (it's a test string), but ensure EICAR queries are run only in lab environments and not as part of general production hunts.
// Follow-ups:
//  1) Cross-reference with DeviceEvents/Alerts or SecurityAlert to see quarantine action and uploaded file evidence (ReportId or AlertId).
//  2) If AV intervened before execution, check scheduled-task registration events to verify persistence attempts succeeded or failed.
//  3) Use these results to validate EDR/AV configuration and telemetry completeness.
let window = 2h;
DeviceFileEvents
| where Timestamp >= ago(window)
| where FileName has "eicar" or FolderPath has "EICARTest" or (FileName has ".com" and FileName has "eicar")
| project Timestamp, DeviceName, FileName, FolderPath, ActionType, InitiatingProcessFileName, ReportId
| order by Timestamp desc
