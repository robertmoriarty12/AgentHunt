--- 1) PowerShell — stealth / encoded commands ---
let timeframe = 1h;
DeviceProcessEvents
| where Timestamp >= ago(timeframe)
| where FileName in~ ("powershell.exe", "pwsh.exe")
| where isnotempty(ProcessCommandLine) and ProcessCommandLine has_any ("-WindowStyle Hidden","-NoProfile","-EncodedCommand","-e ","-nop")
| project Timestamp, DeviceName, FileName, ProcessCommandLine, InitiatingProcessFileName, InitiatingProcessCommandLine
| extend timestamp = Timestamp, AccountCustomEntity = "", HostCustomEntity = DeviceName
| order by Timestamp desc

--- 2) Process inventory — top runners (aggregated) ---
let timeframe = 1h;
DeviceProcessEvents
| where Timestamp >= ago(timeframe)
| summarize
    Count = count(),
    FirstSeen = min(Timestamp),
    LastSeen = max(Timestamp),
    SampleCmdLines = make_set(tostring(ProcessCommandLine), 5),
    SampleInitiators = make_set(tostring(InitiatingProcessFileName), 5)
  by FileName
| order by Count desc
| extend timestamp = FirstSeen, AccountCustomEntity = "", HostCustomEntity = ""

--- 7) RFC5737 TEST-NET probes (lab/test IP ranges) ---
let timeframe = 1h;
DeviceNetworkEvents
| where Timestamp >= ago(timeframe)
| where isnotempty(RemoteIP)
| where ipv4_is_in_range(RemoteIP, "192.0.2.0/24")
   or ipv4_is_in_range(RemoteIP, "198.51.100.0/24")
   or ipv4_is_in_range(RemoteIP, "203.0.113.0/24")
| project Timestamp, DeviceName, InitiatingProcessFileName, InitiatingProcessCommandLine, RemoteIP, RemotePort, ActionType
| extend timestamp = Timestamp, AccountCustomEntity = "", HostCustomEntity = DeviceName
